# MOLI-52 speech enhancement. Not part of requirements.txt on purpose:
# deepfilternet pins numpy<2 and packaging<24, and resemble-enhance pins
# torch==2.1.1. Either pin would downgrade the worker CUDA stack (torch cu128).
#
# DeepFilterNet 3 (MIT), on a machine that already has numpy 2 / torch:
#   pip install --no-deps deepfilternet==0.5.6
#   pip install deepfilterlib==0.5.6 "loguru>=0.5" "appdirs>=1.4"
#
# A fresh venv without numpy 2 can install the pin directly:
#   pip install deepfilternet==0.5.6
#
# Resemble Enhance (MIT) is optional. Install it only where its torch pin
# matches the environment. The worker errors with that explanation if
# audio.enhance.engine is "resemble" and the package is missing.
deepfilternet==0.5.6
deepfilterlib==0.5.6
loguru>=0.5
appdirs>=1.4
